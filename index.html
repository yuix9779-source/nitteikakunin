function doGet() { return HtmlService.createHtmlOutputFromFile('index') .setTitle('空き日程確認') .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } /** * 空きスロットを計算して返すAPI * @param {string} targetDateStr "YYYY-MM-DD" */ function getAvailableSlots(targetDateStr) { const calendar = CalendarApp.getDefaultCalendar(); // 設定: 稼働可能時間（9:00〜18:00）とスロット長（60分） const START_HOUR = 9; const END_HOUR = 18; const SLOT_MINUTES = 60; const targetDate = new Date(targetDateStr); const dayStart = new Date(targetDate.getFullYear(), targetDate.getMonth(), targetDate.getDate(), START_HOUR, 0, 0); const dayEnd = new Date(targetDate.getFullYear(), targetDate.getMonth(), targetDate.getDate(), END_HOUR, 0, 0); // その日のカレンダーイベントを取得 const events = calendar.getEvents(dayStart, dayEnd); // 有効な予定（ToDoを無視）のみを抽出 const busyIntervals = []; events.forEach(event => { const title = event.getTitle().trim().toLowerCase(); // ToDoの判定ルール: タイトル先頭のtodo / [todo] や 終日タスクを無視 const isTodo = title.startsWith('todo') || title.startsWith('[todo]') || title.startsWith('【todo】'); // ToDoではない通常予定のみをブロック対象にする if (!isTodo) { busyIntervals.push({ start: event.getStartTime().getTime(), end: event.getEndTime().getTime() }); } }); // スロットごとに空き状況を判定 const availableSlots = []; let current = new Date(dayStart.getTime()); while (current.getTime() + SLOT_MINUTES * 60 * 1000 <= dayEnd.getTime()) { const slotStart = current.getTime(); const slotEnd = slotStart + SLOT_MINUTES * 60 * 1000; // 既存の予定と重複しているかチェック const isOverlap = busyIntervals.some(busy => { return (slotStart < busy.end && slotEnd > busy.start); }); if (!isOverlap) { const formatTime = d => Utilities.formatDate(new Date(d), 'JST', 'HH:mm'); availableSlots.push(`${formatTime(slotStart)} - ${formatTime(slotEnd)}`); } // 次のスロットへ（30分刻みや60分刻みに変更可能） current = new Date(slotStart + SLOT_MINUTES * 60 * 1000); } return availableSlots; }
